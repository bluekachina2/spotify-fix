cmake_minimum_required(VERSION 3.18)
project(aaudiocapturefix CXX)
set(CMAKE_CXX_STANDARD 17)
set(SRC jni/module.cpp)
option(USE_INLINE_HOOK "also inline-hook libaaudio (needs jni/And64InlineHook.cpp)" OFF)
if(USE_INLINE_HOOK)
  list(APPEND SRC jni/And64InlineHook.cpp)
  add_compile_definitions(USE_INLINE_HOOK)
endif()
add_library(aaudiocapturefix SHARED ${SRC})
target_compile_options(aaudiocapturefix PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
target_link_libraries(aaudiocapturefix log)
